cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(omdrc-pcm-bridge VERSION 0.1.0 LANGUAGES C)
  enable_testing()
else()
  enable_language(C)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(omdrc-pcm-queue STATIC queue.c)
target_compile_options(omdrc-pcm-queue PRIVATE -Wall -Wextra -Werror)
add_executable(pcm-queue-test test_queue.c)
target_compile_options(pcm-queue-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(pcm-queue-test PRIVATE omdrc-pcm-queue)
add_test(NAME pcm-queue COMMAND pcm-queue-test)
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
  find_package(Threads REQUIRED)
  add_executable(omdrc-pcm-bridge bridge.c)
  target_compile_options(omdrc-pcm-bridge PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(omdrc-pcm-bridge PRIVATE omdrc-pcm-queue cuse Threads::Threads)
  install(TARGETS omdrc-pcm-bridge RUNTIME DESTINATION sbin COMPONENT pcm-bridge)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
  add_executable(pcm-oss-test test_oss.c)
  target_compile_options(pcm-oss-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(pcm-oss-test PRIVATE Threads::Threads)
endif()
